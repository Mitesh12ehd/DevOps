head and tail:
    The head and tail commands are Unix/Linux utilities for viewing parts of files:

    head:
        head file.txt           # Shows first 10 lines
        head -n 5 file.txt      # Shows first 5 lines
        head -20 file.txt       # Shows first 20 lines
        head -c 100 file.txt    # Shows first 100 bytes

    tail:
        tail file.txt           # Shows last 10 lines
        tail -n 5 file.txt      # Shows last 5 lines
        tail -20 file.txt       # Shows last 20 lines
        tail -f logfile.txt     # Continuously shows new lines added to file

diff:
    compares files line by line and shows the differences between them

sort :
    organizing data in ascending or descending order based on various criteria

    sort file.txt     # Sort lines alphabetically(lexographically)
    sort file1.txt file2.txt            # Sort combined contents of both files
    sort *.txt                          # Sort all .txt files together
        
    sort with pipe : 
        ls | sort                           # Sort directory listing
        cat file.txt | sort                 # Sort file contents via pipe

    reverse order:  
        sort -r file.txt

    numeric:
        sort -n numbers.txt  
        sort -nr numbers.txt    # numeric and reverse order

wc:
    - word count
    - to count words, character, lines of the file content or output
    
    - Count line number in file    
    	wc file1.txt
    	
    	o/p: 10  50  300 file.txt
    		- lines, words, bytes, filename
    
    - Count lines
    	wc -l file.txt
   	- Count words
   		wc -w file.txt
   	- Count bytes
   		wc -c file.txt
   	- Count char
   		wc -m file.txt
    	
    - Use with other command
    	find /home/mitesh/ -maxdepth 1 -type f -name "*.txt" | wc -l
    
    - count docker containers
    	docker ps -q | wc -l

cut:
    - to get specific part from file content
    - it is useful with 
    	- csv and delimited files
    	- linux configuration files
    	- log files
    	- command output
    	- user and system information
    - command works line by line
    - postion count start from 1 index
	
	- get first char from each line
		cut -c 1 file.txt
	
    - get first 5 character from each line
        cut -c 1-5 filename.txt
    
    - Characters from position 3 onwards
    	cut -c 3- file.txt
    	
    - First 3 characters and character 6
    	cut -c 1-3,6 file.txt

    - get specific character:
        cut -c 1,3,5 original.txt

    - extract by delimiter and field(-d and -f)
    	- we can also delimate by space ' '

        Suppose /etc/passwd contains:
            root:x:0:0:root:/root:/bin/bash
            ubuntu:x:1000:1000:Ubuntu:/home/ubuntu:/bin/bash

        Get usernames:
            cut -d ':' -f 1 /etc/passwd

            -f1 extract first field

            Output:
                root
                ubuntu

        Get Home Directories
            cut -d ':' -f 6 /etc/passwd

            Output:
                /root
                /home/ubuntu

        Extract Multiple Fields:

            file: data.csv
                John,25,Ahmedabad
                Alice,30,Mumbai

            cut -d ',' -f 1,3 data.csv
            cut -d ',' -f 1-3 data.csv       # to get range of field
            
	- Extract process IDs
		ps -ef | tr -s ' ' | cut -d ' ' -f 2

nohup:
	- no hang up
	- used to run a command or process that continues executing even after the user logs out or the terminal session is closed.
    
    Without nohup:
    	- ./backup.sh
    	- normally shell sends SIGNUP singla to jobs when session ends, this can cause process to terminate
   
   	With nohup:
   		- to prevent this, we use nohup
   		- nohup ./backup.sh 
   		- This makes the command ignore the SIGHUP signal.
   		- Add & to run in background
   		
   	Syntax:
   		nohup command
   		
   	Where nohup save the output?
   		- it redirect standard output and standard error to file names nohup.out
   		- file is present in the directory from where you ran command
   		
   	Redirect output to a custom log file
   		nohup ./backup.sh > backup.log 2>&1 &
   		
   		- Monitor the log continuesly:
   			tail -f backup.log
    
    - it used to store output of command 
    - it also append output
         
    ex.
        nohup free -h 
        # it create nohup.out file
